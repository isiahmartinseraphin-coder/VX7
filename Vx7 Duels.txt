local function new(class, props, children)
	local o = Instance.new(class)
	for k, v in pairs(props or {}) do o[k] = v end
	for _, c in ipairs(children or {}) do c.Parent = o end
	return o
end
local function corner(r) return new("UICorner", { CornerRadius = UDim.new(0, r) }) end
local function stroke(c, t, trans)
	return new("UIStroke", {
		Color = c or Color3.fromRGB(45, 75, 125),
		Thickness = t or 1,
		Transparency = trans or 0.4,

end
local function pad(l, r, t, b)
	return new("UIPadding", {
		PaddingLeft = UDim.new(0, l or 0), PaddingRight = UDim.new(0, r or 0),
		PaddingTop = UDim.new(0, t or 0), PaddingBottom = UDim.new(0, b or 0)
	})
end

	if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

		listeningBtn.btn.Text = listeningBtn.prev
		listeningBtn.btn.TextColor3 = COL.text
		listeningBtn = nil
		return
	end
	if listeningBtn then listeningBtn.onKey(input.KeyCode); listeningBtn = nil; return end

	if cb then cb() end
end)

local function buildMovement()
	sectionHeader("SPEED").Parent = body
	makeMethodSelector().Parent = body
	local opts = {
		{ name = "Normal Speed", desc = "default mode", key = "Q", normal = 57, steal = 27, selected = true },
		{ name = "Lagger Speed", desc = "use against lagger", key = "Z", normal = 39, steal = 20, selected = false },
		{ name = "Custom Speed", desc = "use if Custom spd", key = "+", normal = 67, steal = 34, selected = false },
	}
	local radios = {}
	for _, o in ipairs(opts) do
		local row, radio = makeSpeedOption(o)
		row.Parent = body
		table.insert(radios, { row = row, radio = radio })
	end
	for i, item in ipairs(radios) do
		item.radio.MouseButton1Click:Connect(function()
			for j, other in ipairs(radios) do
				local sel = (j == i)
				other.radio.BackgroundColor3 = COL.inputBg
				other.radio.BackgroundTransparency = 0.3
				for _, ch in ipairs(other.radio:GetChildren()) do
					if ch:IsA("Frame") then ch:Destroy() end
				end
				local rStroke = other.radio:FindFirstChildOfClass("UIStroke")
				if rStroke then
					rStroke.Color = sel and COL.blue or COL.border
					rStroke.Transparency = sel and 0 or 0.35
					rStroke.Thickness = sel and 1.8 or 1.5
				end
				if sel then
					new("Frame", {
						Parent = other.radio, Name = "Dot",
						Size = UDim2.fromOffset(10, 10), Position = UDim2.fromOffset(6, 6),
						BackgroundColor3 = COL.blue, BorderSizePixel = 0,
					}, { corner(5) })
				end
				other.row.BackgroundColor3 = sel and COL.panelAlt or COL.panel
				local s = other.row:FindFirstChildOfClass("UIStroke")
				if s then
					s.Color = sel and COL.blue or COL.border
					s.Transparency = sel and 0.15 or 0.4
				end
			end
		end)
	end
	sectionHeader("SPEED KEYS").Parent = body
	makeSpeedKeys().Parent = body
end
